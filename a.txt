---------------------------------------------------------
-- スプーファーのロジックコア部分
---------------------------------------------------------

local function casepatt(pattern)
    return string.gsub(pattern, "(%%?)(.)", function(percent, letter)
        if percent ~= "" or not string.match(letter, "%a") then
            return percent .. letter
        else
            return string.format("[%s%s]", string.lower(letter), string.upper(letter))
        end
    end)
end

-- 通常のGUIテキスト置換
local function replace(item)
    if new_name == "" then return end
    
    for _, v in pairs(replaces_str) do
        for _, t in pairs({ "Text", "Message", "ToolTip", "Value" }) do
            pcall(function()
                if string.find(item[t], v, nil, true) then
                    item[t] = string.gsub(item[t], v, new_name)
                elseif string.find(item[t], string.lower(v), nil, true) then
                    item[t] = string.gsub(item[t], string.lower(v), string.lower(new_name))
                elseif string.find(item[t], string.upper(v), nil, true) then
                    item[t] = string.gsub(item[t], string.upper(v), string.upper(new_name))
                elseif string.find(string.lower(item[t]), string.lower(v), nil, true) then
                    item[t] = string.gsub(item[t], casepatt(v), new_name)
                end
            end)
        end
        
        if rename_instances then
            pcall(function()
                if string.find(item.Name, v, nil, true) then
                    item.Name = string.gsub(item.Name, v, new_name)
                elseif string.find(item.Name, string.lower(v), nil, true) then
                    item.Name = string.gsub(item.Name, string.lower(v), string.lower(new_name))
                elseif string.find(item.Name, string.upper(v), nil, true) then
                    item.Name = string.gsub(item.Name, string.lower(v), string.upper(new_name))
                elseif string.find(string.lower(item.Name), string.lower(v), nil, true) then
                    item.Name = string.gsub(item.Name, casepatt(v), new_name)
                end
            end)
        end
    end
end

-- 【新規】Rivalsのキルログシステム専用ハック
local function handleRivalsKillLog(child)
    if new_name == "" then return end
    
    -- RivalsのキルログUI要素（KillFeed、Row、Templateなど）にあなたの名前が含まれているかチェック
    for _, v in pairs(replaces_str) do
        -- 子要素すべての文字列（テキストラベルや属性値など）を網羅的に強制スキャンして上書き
        for _, desc in ipairs(child:GetDescendants()) do
            replace(desc)
        end
        
        -- キルログのデータ属性やオブジェクト名自体に名前が組み込まれている場合も対応
        pcall(function()
            if string.find(child.Name, v, nil, true) then
                child.Name = string.gsub(child.Name, v, new_name)
            end
        end)
    end
end

local function scan_and_replace()
    if new_name == "" then return end
    local scan_que = { CoreGui:GetDescendants(), PlayerGui:GetDescendants(), workspace:GetDescendants() }
    local last_break = 0
    
    for _, items in ipairs(scan_que) do
        for _, gui in ipairs(items) do
            last_break = 1 + last_break
            if last_break >= 6000 then
                RunService.RenderStepped:Wait()
                last_break = 0
            end
            replace(gui)
        end
    end
end

-- 画面へのUI追加を監視
CoreGui.DescendantAdded:Connect(function(child)
    replace(child)
    handleRivalsKillLog(child)
end)

PlayerGui.DescendantAdded:Connect(function(child)
    replace(child)
    handleRivalsKillLog(child)
end)

workspace.DescendantAdded:Connect(replace)

-- 毎フレームのループで常に自分の名前とキャラクター名の偽装を維持
RunService.RenderStepped:Connect(function()
    if new_name == "" then return end
    
    if rename_instances then
        pcall(function()
            LocalPlayer.Name = new_name
            if LocalPlayer.Character then
                LocalPlayer.Character.Name = new_name
            end
        end)
    end
    pcall(replace, workspace)
    
    -- 【追加】地上のキルログフレームを毎フレーム常時監視して強制上書き
    pcall(function()
        for _, gui in ipairs(PlayerGui:GetDescendants()) do
            if string.find(string.lower(gui.Name), "kill") or string.find(string.lower(gui.Name), "feed") then
                for _, textLabel in ipairs(gui:GetDescendants()) do
                    if textLabel:IsA("TextLabel") or textLabel:IsA("TextBox") then
                        replace(textLabel)
                    end
                end
            end
        end
    end)
end)

-- UIボタンの処理
ApplyButton.MouseButton1Click:Connect(function()
    local input_text = NameInput.Text
    if input_text and input_text ~= "" then
        new_name = tostring(input_text)
        ApplyButton.Text = "変更完了！"
        ApplyButton.BackgroundColor3 = Color3.fromRGB(0, 120, 200)
        
        task.spawn(scan_and_replace)
        
        task.wait(1)
        ApplyButton.Text = "名前を変更 (Apply)"
        ApplyButton.BackgroundColor3 = Color3.fromRGB(0, 150, 100)
    end
end)
